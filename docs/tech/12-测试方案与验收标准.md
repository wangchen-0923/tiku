### 1.3 测试数据原则

1. **可用固定 seed 复现**：所有随机路径（组卷、洗牌、乱序、练习生成）必须显式传 `rng`，禁止依赖 `random.SystemRandom()`（见 §2.3 的 `TC-PAP-102`）。
2. **绝不使用生产数据**：脱敏测试用的「脏数据」在 `tests/fixtures/dirty_pii.json` 里人工构造，不含真实学生信息。
3. **专项数据集有版本号**：`omr_dataset_v1` / `figure_dataset_v1`，基线随版本记录，换数据集必须升版本并重设基线。

### 2.2 ★关键代码示例一：乱序映射（一人一卷）

> 设计文档 2.3：**「扫描时读到的『第 7 题涂了 B』，必须用该学生的 `question_order` 映射回实际题目。如果映射错了，全班成绩全错。」**

```python
# server/apps/omr/tests/test_question_order_mapping.py
import pytest
from apps.assignment.models import AssignmentStudent
from apps.grading.mapping import resolve_paper_item, resolve_option_answer


@pytest.fixture
def shuffle_env(db, assignment_of_45_students):
    """发布任务开启题目乱序 + 选项乱序，逐生持久化映射。"""
    a = assignment_of_45_students
    a.shuffle_questions = True
    a.shuffle_options = True
    a.save()
    publish_assignment(a)                       # 生成 assignment_student.question_order
    return a


def test_card_seq_maps_to_own_paper_item(shuffle_env):
    """卡上第 7 题 → 该生的实际 paper_item（不是试卷的第 7 题）。"""
    stu = shuffle_env.students.get(student_no="20260312")
    order = stu.question_order                       # [paper_item_id, ...]
    assert len(order) == shuffle_env.paper.item_count
    got = resolve_paper_item(assignment=shuffle_env, student=stu, card_seq=7)
    assert got.id == order[6]


def test_different_students_do_not_cross(shuffle_env):
    """45 人互不串号：同一卡面题号在不同学生上映射到不同 paper_item。"""
    seen = {}
    for stu in shuffle_env.students.all():
        for seq in range(1, shuffle_env.paper.item_count + 1):
            pid = resolve_paper_item(assignment=shuffle_env, student=stu, card_seq=seq).id
            seen.setdefault(seq, set()).add(pid)
    # 只要题库足够大，同一题号不应在所有人身上都是同一道题
    assert any(len(v) > 1 for v in seen.values()), "所有学生映射完全一致，乱序未生效"
    # 每个学生自己的映射必须是双射
    for stu in shuffle_env.students.all():
        assert len(set(stu.question_order)) == len(stu.question_order)


def test_mapping_is_persisted_not_regenerated(shuffle_env):
    """映射不可复现则报错：两次读取必须得到同一结果（禁止读时重新随机）。"""
    stu = shuffle_env.students.first()
    first = list(stu.question_order)
    stu.refresh_from_db()
    assert stu.question_order == first
    # 重新发布同一任务不得改变已发布学生的映射
    with pytest.raises(AlreadyPublished):
        publish_assignment(shuffle_env)


def test_grading_uses_student_option_order(shuffle_env):
    """选项乱序：学生涂 B 必须按**该学生的** option_order 还原成原题选项标识。"""
    stu = shuffle_env.students.get(student_no="20260313")
    pid = stu.question_order[6]
    item = PaperItem.objects.get(id=pid)
    opt_order = stu.option_order[str(pid)]           # 例如 ["C", "A", "D", "B"]
    # 卡上涂了第 2 个位置 → 实际是原题的 opt_order[1] 选项
    assert resolve_option_answer(item=item, option_order=opt_order, card_pos=2) == opt_order[1]
    # 反例：用别的学生的 option_order 会得到不同结果（证明必须逐生）
    other = shuffle_env.students.exclude(id=stu.id).first()
    if str(pid) in (other.option_order or {}):
        assert resolve_option_answer(item=item, option_order=other.option_order[str(pid)],
                                     card_pos=2) != opt_order[1] or True   # 允许巧合，不强制不等


def test_missing_mapping_raises(shuffle_env):
    """映射缺失必须显式报错，绝不 fallback 到卷面顺序。"""
    stu = shuffle_env.students.first()
    AssignmentStudent.objects.filter(id=stu.id).update(question_order=None)
    with pytest.raises(QuestionOrderMissing):
        resolve_paper_item(assignment=shuffle_env, student=stu, card_seq=7)
```

### 2.3 ★关键代码示例二：锚点校正防线（防整卡错答案）

> 设计文档 9.3.2③：**「最坏的情况是恰好每个位置都有一个涂点被判定为『已涂』，系统就会输出一整张看似正常但完全错误的答案。回投影校验能拦住这种情况。」**

```python
# server/apps/omr/tests/test_anchor_defense.py
import numpy as np
import pytest
from apps.omr.engine import detect_anchors, rectify_and_validate
from apps.omr.exceptions import AnchorNotFound, AnchorDeviationTooLarge


def test_back_projection_blocks_wrong_anchor(synthetic_sheet, template_v1):
    """
    构造「锚点识别错误」的输入：把印刷边框角当成锚点（整体偏移）。
    断言：回投影偏差 > 页面宽度 1% → 抛错 → 整卡进人工队列，绝不输出答案。
    """
    img = synthetic_sheet.render()                                   # 正确锚点的卡
    wrong_anchors = synthetic_sheet.anchors_shifted(dx=0.02, dy=0.01) # 偏移 2% 页宽
    with pytest.raises(AnchorDeviationTooLarge):
        rectify_and_validate(img, wrong_anchors, template_v1)


def test_correct_anchor_passes(synthetic_sheet, template_v1):
    img = synthetic_sheet.render()
    warped, quality = rectify_and_validate(img, synthetic_sheet.anchors(), template_v1)
    assert quality >= 0.8
    assert warped.shape[0] == template_v1.pixel_size_at_dpi(300)[1]


def test_missing_anchor_goes_to_manual_queue(scan_service, missing_anchor_image):
    """锚点缺失（20 张测试集中的样本）→ SCAN.ANCHOR_NOT_FOUND，卡状态 NEEDS_REVIEW。"""
    card = scan_service.ingest(missing_anchor_image)
    assert card.status == "NEEDS_REVIEW"
    assert card.fail_reason and "ANCHOR" in card.fail_reason.upper()


def test_all_positions_filled_anomaly_is_flagged(scan_service, all_filled_regular_image):
    """已涂题数 = 100% 且答案呈规律（全 A / ABCD 循环）→ 整卡可疑，进仲裁。"""
    card = scan_service.ingest(all_filled_regular_image)
    assert card.status == "NEEDS_REVIEW"
    assert any("REGULAR_PATTERN" in f for f in card.uncertain_items or [])


def test_underspecified_thresholds_per_channel():
    """手机与扫描仪必须两套阈值（附录 F：网页 0.45/0.15，手机 0.50/0.20）。"""
    from apps.omr.config import thresholds_for
    web = thresholds_for(channel="UPLOAD")
    phone = thresholds_for(channel="PHONE_H5")
    assert (web.filled, web.empty) == (0.45, 0.15)
    assert (phone.filled, phone.empty) == (0.50, 0.20)
    assert phone.filled > web.filled            # 手机更严，防误判
```

### 2.4 单元测试的固定种子与统计检验

```python
# server/apps/paper/tests/test_fisher_yates.py
import random
from collections import Counter
from apps.paper.shuffle import fisher_yates_shuffle


def test_position_distribution_is_uniform():
    """100 万次洗牌后位置分布均匀性统计检验（16.4 强制用例）。"""
    n, trials = 10, 200_000                 # 10 个元素 × 20 万次 = 200 万次放置
    rng = random.Random(20261001)
    counts = [[Counter() for _ in range(n)] for _ in range(n)]  # counts[pos][value]
    for _ in range(trials):
        arr = fisher_yates_shuffle(list(range(n)), rng)
        for pos, value in enumerate(arr):
            counts[pos][value] += 1
    expected = trials / n
    # 卡方检验：自由度 n*(n-1)，取 α = 0.001（避免 CI 偶发红灯）
    chi2 = sum((counts[p][v] - expected) ** 2 / expected
               for p in range(n) for v in range(n))
    critical = chi2_critical(df=n * (n - 1), alpha=0.001)
    assert chi2 < critical, f"分布不均匀 chi2={chi2:.1f} > {critical}"


def test_requires_explicit_rng():
    """禁止依赖 SystemRandom 的可复现路径（对应 A11「同 seed 可复现」）。"""
    import inspect
    from apps.paper.shuffle import fisher_yates_shuffle as f
    sig = inspect.signature(f)
    assert sig.parameters["rng"].default is inspect.Parameter.empty, \
        "fisher_yates_shuffle 必须强制传入 rng（设计文档 8.2 的默认 SystemRandom 与 A11 冲突）"
```

> **统计检验的确定性要求（重要）**：卡方检验天生有 Type I error。为让 CI 稳定，必须：① 用固定 `rng` 种子（检验本身因此**完全确定**，不抖动）；② α 取 0.001；③ 断言只报「严重偏离」，一致性检查（同 seed 两次结果相同）单独用一个确定性用例。

### 3.2 测试执行脚本骨架

```python
# scripts/omr_eval.py
"""
用法：
  python scripts/omr_eval.py --dataset datasets/omr_dataset_v1 \
      --channel PHONE_H5 --out artifacts/omr_phone.json --baseline baselines/omr_phone_v1.json
"""
from __future__ import annotations
import argparse, json, statistics
from pathlib import Path
from apps.omr.engine import recognize_card
from apps.omr.config import thresholds_for, THRESHOLD_VERSION


def load_ground_truth(dataset_dir: Path, channel: str) -> list[dict]:
    cases = []
    for yml in sorted((dataset_dir / "labels").glob("*.yaml")):
        meta = yaml.safe_load(yml.read_text(encoding="utf-8"))
        if meta["channel"] != channel:
            continue
        meta["image"] = dataset_dir / "images" / meta["file"]
        cases.append(meta)
    return cases


def evaluate(cases: list[dict]) -> dict:
    tp = fp = fn = tn = 0                 # 涂点级混淆矩阵（已涂=正）
    cards_silent = cards_total = 0
    no_ok = no_total = 0
    anchor_fail = 0
    fail_reasons: dict[str, int] = {}
    per_card = []

    for case in cases:
        cards_total += 1
        result = recognize_card(case["image"], template_version=case["template_version"],
                                channel=case["channel"])
        if result.status == "FAILED":
            anchor_fail += 1
            fail_reasons[result.fail_reason] = fail_reasons.get(result.fail_reason, 0) + 1
            continue

        # ① 学号
        if case.get("student_no"):
            no_total += 1
            no_ok += int(result.student_no_text == case["student_no"])

        # ② 逐题混淆矩阵（只统计该卡实际涂的与判定的）
        card_wrong = 0
        for seq_s, expected in case["answers"].items():
            got = (result.answers.get(int(seq_s)) or {}).get("value")
            for label in case["option_labels"][seq_s]:
                exp_filled = _is_expected_filled(expected, label)
                got_filled = _is_got_filled(got, label)
                tp += exp_filled and got_filled
                tn += (not exp_filled) and (not got_filled)
                fp += (not exp_filled) and got_filled
                fn += exp_filled and (not got_filled)
            if _norm(got) != _norm(expected):
                card_wrong += 1

        silent = (result.status == "AUTO_OK" and card_wrong == 0)
        cards_silent += int(silent)
        per_card.append({"file": case["file"], "silent": silent,
                         "wrong_items": card_wrong,
                         "student_no_ok": result.student_no_text == case.get("student_no"),
                         "confidence": result.overall_confidence})

    total_points = tp + fp + fn + tn or 1
    return {
        "threshold_version": THRESHOLD_VERSION,
        "channel": cases[0]["channel"] if cases else None,
        "n_cards": cards_total,
        "item_accuracy": (tp + tn) / total_points,
        "miss_rate": fn / (tp + fn or 1),          # 漏检率（已涂判未涂）
        "false_fill_rate": fp / (fp + tn or 1),    # 误涂率（未涂判已涂）
        "silent_pass_rate": cards_silent / (cards_total or 1),
        "student_no_accuracy": no_ok / (no_total or 1),
        "anchor_fail_rate": anchor_fail / (cards_total or 1),
        "fail_reasons": fail_reasons,
        "per_card": per_card,
    }


def compare_and_gate(report: dict, baseline: dict, *, family: str) -> int:
    """回归门禁：低于基线即阻断合并（15.5）。返回 0 = 通过，1 = 阻断。"""
    checks = [
        ("silent_pass_rate", "ge"),
        ("item_accuracy", "ge"),
        ("student_no_accuracy", "ge"),
        ("miss_rate", "le"),
        ("false_fill_rate", "le"),
    ]
    failed = []
    for key, direction in checks:
        cur, base = report[key], baseline[key]
        ok = cur >= base if direction == "ge" else cur <= base
        print(f"{'OK ' if ok else 'FAIL'} {family}.{key}: {cur:.4f} (baseline {base:.4f})")
        if not ok:
            failed.append(key)
    if failed:
        print(f"\n[BLOCKED] OMR 回归未达标：{failed}")
        return 1
    return 0


if __name__ == "__main__":
    ap = argparse.ArgumentParser()
    ap.add_argument("--dataset", required=True)
    ap.add_argument("--channel", required=True, choices=["PHONE_H5", "UPLOAD"])
    ap.add_argument("--out", required=True)
    ap.add_argument("--baseline")
    args = ap.parse_args()

    report = evaluate(load_ground_truth(Path(args.dataset), args.channel))
    Path(args.out).write_text(json.dumps(report, ensure_ascii=False, indent=2))
    print(json.dumps({k: v for k, v in report.items() if k != "per_card"},
                     ensure_ascii=False, indent=2))
    if args.baseline:
        raise SystemExit(compare_and_gate(
            report, json.loads(Path(args.baseline).read_text()),
            family=f"omr_{args.channel.lower()}"))
```

### 4.2 指标计算脚本骨架

```python
# scripts/figure_eval.py
"""
用法：
  python scripts/figure_eval.py --dataset datasets/figure_dataset_v1 \
      --out artifacts/figure_v1.json --baseline baselines/figure_v1.json
指标（7.4.6）：召回率 / 误报率 / 归属正确率 / 裁切完整率 / 公式转 LaTeX 可用率 / 校对耗时
"""
from __future__ import annotations
import argparse, json, math
from pathlib import Path
from apps.ingest.pipeline import run_ingest_for_dataset_file


def bbox_iou(a, b) -> float:
    ax1, ay1, ax2, ay2 = a; bx1, by1, bx2, by2 = b
    ix1, iy1 = max(ax1, bx1), max(ay1, by1)
    ix2, iy2 = min(ax2, bx2), min(ay2, by2)
    iw, ih = max(0.0, ix2 - ix1), max(0.0, iy2 - iy1)
    inter = iw * ih
    union = (ax2-ax1)*(ay2-ay1) + (bx2-bx1)*(by2-by1) - inter
    return inter / union if union > 0 else 0.0


def evaluate(cases: list[dict]) -> dict:
    tp = fp = fn = 0
    binding_ok = binding_total = 0
    crop_ok = crop_total = 0
    formula_total = formula_latex_ok = 0
    suspect_ok = suspect_total = 0
    per_case = []

    for case in cases:
        result = run_ingest_for_dataset_file(case)      # 走与线上完全相同的流水线
        got = result.figures
        exp = case["expected_figures"]

        # ── 一对多匹配：IoU > 0.7 视为同一张图 ──
        matched_exp, matched_got = set(), set()
        for i, e in enumerate(exp):
            for j, g in enumerate(got):
                if g.src_page == e["page"] and bbox_iou(g.bbox_rel, e["bbox_rel"]) > 0.7:
                    matched_exp.add(i); matched_got.add(j)
                    # 归属正确率
                    binding_total += 1
                    exp_belong = e["belongs_to_seq"]
                    exp_set = set(exp_belong if isinstance(exp_belong, list) else [exp_belong])
                    got_set = set(g.belongs_to_seq if isinstance(g.belongs_to_seq, list)
                                  else [g.belongs_to_seq])
                    binding_ok += int(exp_set == got_set)
                    # 裁切完整率
                    crop_total += 1
                    crop_ok += int(float(g.quality_score or 0) >= 0.5
                                   and "SUSPECT_CROP" not in (g.quality_flags or []))
                    # SUSPECT_CROP 期望值核对
                    if e.get("touch_edge") is not None:
                        suspect_total += 1
                        suspect_ok += int(("SUSPECT_CROP" in (g.quality_flags or []))
                                          == bool(e["touch_edge"]))
                    break
        tp += len(matched_exp)
        fn += len(exp) - len(matched_exp)
        fp += len(got) - len(matched_got)

        # ── 公式：不得被裁为图片 ──
        for ef in case.get("expected_formulas", []):
            formula_total += 1
            mis_cropped = any(bbox_iou(g.bbox_rel, ef["bbox_rel"]) > 0.7 for g in got)
            latex_ok = any(bbox_iou(l.bbox_rel, ef["bbox_rel"]) > 0.7
                           and looks_compilable(l.latex) for l in result.formulas)
            formula_latex_ok += int((not mis_cropped) and latex_ok)

        per_case.append({"file": case["file"], "expected": len(exp), "got": len(got),
                         "matched": len(matched_exp)})

    recall = tp / (tp + fn) if (tp + fn) else 0.0
    false_positive = fp / (tp + fp) if (tp + fp) else 0.0
    return {
        "n_files": len(cases),
        "figure_recall": recall,
        "false_positive_rate": false_positive,
        "binding_accuracy": binding_ok / (binding_total or 1),
        "crop_integrity": crop_ok / (crop_total or 1),
        "formula_to_latex_usable": formula_latex_ok / (formula_total or 1),
        "formula_mis_crop_rate": 1 - (formula_latex_ok / (formula_total or 1)),
        "suspect_crop_detection_accuracy": suspect_ok / (suspect_total or 1),
        "per_case": per_case,
    }


# 校对耗时：需要真人计时，单独记录（见 §4.3 的记录表）
CORRECT_TIME_SCHEMA = {
    "file": "str", "teacher": "str", "figures": "int",
    "confirm_seconds_total": "float", "per_figure_seconds": "float",
    "used_batch_confirm": "bool", "suspicious_speed": "bool",
}
```

### 4.3 ★「沉默错误率」的实验设计

> 设计文档 7.4.6：**「沉默错误率是错误图（裁错/归属错/丢失）中，教师未发现而确认入库的比例。这是这个项目里我最看重的一个指标，也是最难测的一个。」**

#### 4.3.1 实验设计（v2.1 修正版）

设计文档 16.3 只给了「3 位教师各自正常校对 20 份试卷，另 1 人逐题核对」。为了让数字**可解释**，必须补三个前提：

| 前提 | 做法 | 为什么必须 |
| --- | --- | --- |
| **① 先建立机器 ground truth** | 由标注负责人（非参与实验的教师）对同一批 20 份试卷逐题标注期望图集合（§4.1 的 `expected_figures`），**在教师校对之前完成并封存** | 否则「总错误数」的分母不可知：丢失的图不会产生任何记录，教师根本无从确认，指标会漏掉最严重的一类错误 |
| **② 教师按正常流程操作，不做任何提示** | 使用真实界面（逐图确认 + 键盘流），不告知「本实验在测漏检」 | 提示会改变行为，指标失效 |
| **③ 比对的是「最终入库图集合」而不是「教师点击过的图」** | 实验结束后导出该批次全部 `question_image` 记录（含 `deleted_at` 为空者），与 ground truth 逐张比对 | 只有落到库里的图才会进试卷 |

**样本量**：**3 位教师 × 20 份 = 60 份**（设计文档 16.3 的写法），每份自选 6–10 张图的试卷。同时记录**第 4 人**（标注负责人）的逐题核对结论。

#### 4.3.2 指标定义（避免歧义）

```text
总错误数 TE = 裁切错误数 + 归属错误数 + 丢失图数 + 误报图数（与 ground truth 比对得出）
教师发现数 FD = 教师在系统中做出修正（RECROP / REPLACE / MANUAL_CAPTURE / DELETED）或标记存疑的图数
沉默错误数 SE = TE − FD（教师未发现的错误）
沉默错误率 SER = SE / max(TE, 1)

配套指标：
  净入库错误率 = 最终入库的错误图数 / ground truth 图总数      ← 与 A7 的口径对齐
  确认有效率 = FD / (FD + SE)，反映「强制确认」是否真的有效（对应 R21）
```

#### 4.3.4 实验的两个已知局限（必须写进报告）

1. **「丢失图」类的 SER 结构性偏高**：丢失的图不产生 `question_image` 记录，界面上没有任何提示，教师**不可能**发现。因此建议把 SER 拆成两个子指标：
   - `SER_可发现` = 裁切错 + 归属错的沉默率（这类教师本可以发现，衡量确认有效性）；
   - `SER_不可发现` = 丢失图的沉默率（这类**只能靠召回率指标**控制，不能算在教师头上）。
2. **实验环境与真实环境的差异**：实验中的 60 份试卷教师不熟悉，注意力高于日常。真实 SER 通常**高于**实验值。因此报告必须注明「实验值为下界」。

```python
# server/tests/integration/test_closed_loop.py
@pytest.mark.django_db(transaction=True)
def test_full_closed_loop(seed_org, celery_eager, api_client, teacher_token):
    """一条测试跑完五环节，每环断言输入输出。任何一环改动都会在这里暴露。"""
    # ① 组卷
    paper = api_post(api_client, "/api/v1/papers/171/auto-compose/", token=teacher_token,
                     {"template": TEMPLATE_23, "seed": 20261001})
    assert paper["item_count"] == 23 and paper["status"] == "DRAFT"

    # ①′ 绑定答题卡（校验必须通过）
    binding = api_post(api_client, f"/api/v1/papers/{paper['id']}/sheet-binding/",
                       token=teacher_token, {"sheet_template_id": 3})
    assert binding["is_primary"] is True

    # ①″ 配图关口二：全部确认后定稿
    verify = api_get(api_client, f"/api/v1/papers/{paper['id']}/figure-verify/", token=teacher_token)
    if verify["pending_count"]:
        api_post(api_client, f"/api/v1/papers/{paper['id']}/figure-verify/confirm-all/",
                 token=teacher_token)
    assert api_post(api_client, f"/api/v1/papers/{paper['id']}/finalize/",
                    token=teacher_token)["status"] == "FINALIZED"

    # ② 发布
    asg = api_post(api_client, "/api/v1/assignments/", token=teacher_token,
                   {"paper_id": paper["id"], "classroom_id": 4, "channel": "BOTH",
                    "sheet_template_id": 3, "shuffle_questions": True, "shuffle_options": True})
    api_post(api_client, f"/api/v1/assignments/{asg['id']}/publish/", token=teacher_token)
    students = AssignmentStudent.objects.filter(assignment_id=asg["id"])
    assert students.count() == 45
    assert all(len(s.question_order) == 23 for s in students)

    # ③ 作答 + ③′ 扫描
    for s in students[:20]:
        submit_online(asg, s, answers=ONLINE_ANSWERS)
    for s in students[20:]:
        ingest_scan_card(asg, s, image=PHONE_CARD_SAMPLE, channel="PHONE_H5")

    # ④ 批改
    api_post(api_client, f"/api/v1/grading/assignments/{asg['id']}/auto-grade/", token=teacher_token)
    assert ScoreResult.objects.filter(assignment_id=asg["id"]).count() == 45 * 23
    enter_subjective_scores(asg, per_item=True)

    # ④′ 修正重算
    with assert_num_queries_under(120):
        api_patch(api_client, f"/api/v1/grading/scores/{first_score_id}/", token=teacher_token,
                  {"score": 3.5})
    assert_recomputed_within(seconds=30, assignment_id=asg["id"])

    # ⑤ 学情
    report = api_get(api_client, f"/api/v1/analytics/classrooms/4/heatmap/", token=teacher_token)
    assert report["nodes"] and all("mastery" in n for n in report["nodes"])
    top = api_get(api_client, f"/api/v1/analytics/classrooms/4/top-errors/", token=teacher_token)
    assert all("option_distribution" in q for q in top["items"])

    # ⑥ 练习
    ps = api_post(api_client, "/api/v1/practice/generate/", token=teacher_token,
                  {"source_assignment_id": asg["id"], "student_ids": [s.student_id for s in students]})
    run_task(ps["task_id"])                      # eager
    sets = PracticeSet.objects.filter(source_assignment_id=asg["id"])
    assert sets.count() == 45
    src_sections = Paper.objects.get(id=paper["id"]).section_structure()
    for ps_ in sets:
        assert ps_.paper.section_structure() == src_sections      # A24 结构一致
```

### 9.2 压测数据准备（用 seed 生成 2 万题的库）

```bash
# 1) 机构与基础数据（对应 17.3 的 seed 要求）
python manage.py seed --profile=perf --orgs=1 --classrooms=8 --students=400

# 2) 题库：2 万题（含 30% 含图，覆盖 500 个知识点 × 5 个难度档位）

  --org=1 --subject=physics --grade-band=SENIOR \
  --questions=20000 --figure-ratio=0.3 --nodes=500 \
  --difficulty-dist=1:0.15,2:0.25,3:0.30,4:0.20,5:0.10
```

**性能测试的执行顺序（重要）**：先跑**基线压测**（只在本地/staging，无其他负载）→ 记录基线 → 再跑「混合负载压测」（模拟真实：40% 题库浏览 + 30% 扫描 + 20% 组卷 + 10% 学情）。**只看基线不看混合负载会漏掉队列争抢问题**（这正是 4.4 队列分级的价值）。

## 11. 缺陷与质量门禁

### 11.1 缺陷分级标准

| 级别 | 定义 | 典型示例 | 响应时限 | 修复时限 | 发布影响 |
| --- | --- | --- | --- | --- | --- |
| **P0 阻断** | 数据错误或教学事故；安全/合规问题；主流程不可用 | 乱序映射错误导致全班成绩错；学生数据未脱敏外发；跨机构越权；配图错配进入试卷；成绩记到别人头上 | **即刻（15 分钟内响应）** | 当日修复 + 热修 | **禁止发布**；已发布需回滚 |
| **P1 严重** | 核心功能不可用但有绕过手段；指标不达标 | OMR 静默通过率低于基线；AI 生题可绕过 DRAFT；导出失败；扫描批次卡死 | 1 小时内 | 2 个工作日内 | 需 PM 决策，默认不发布 |
| **P2 一般** | 非核心功能异常或体验明显受损 | 配图确认键盘流某个键失效；热力图配色错；报告导出偶发超时 | 当日内 | 本迭代内 | 可发布（列入已知问题） |
| **P3 轻微** | 文案/样式/低频边界 | 空状态文案不清晰；某个提示语有错别字 | 3 日内 | 下个迭代 | 可发布 |

**判定补充规则**：

1. **「沉默错误」类问题一律至少 P1**（裁错图、归属错、丢失图进入题库）——因为教师很难发现（设计文档 7.4.1）。
2. **涉及学生身份数据的任何泄漏一律 P0**，不分数据量大小。
3. **可测性缺陷也算缺陷**：如果某个指标无法测量（见 §14），提 P2 并挂到「测试可行性」跟踪。
